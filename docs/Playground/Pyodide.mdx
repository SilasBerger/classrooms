```py live_pyo id=97d2f8ba-4d19-4481-8460-87ee7ec826fc title=Notenrechner
user_input = input("Note eingeben oder 'ende' zum Beenden: ")
noten = []
while user_input.lower() != "ende":
    try:
        note = float(user_input)
        if 1.0 <= note <= 6.0:
            noten.append(note)
        else:
            print("Ungültige Note. Bitte eine Zahl zwischen 1.0 und 6.0 eingeben.")
    except ValueError:
        print("Ungültige Eingabe. Bitte eine Zahl eingeben.")
    user_input = input("Note eingeben oder 'ende' zum Beenden: ")

durchschnitt = sum(noten) / len(noten) if noten else 0
print(f"Durchschnittsnote: {durchschnitt:.2f}")
```
